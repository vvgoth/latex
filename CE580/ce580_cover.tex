clear; clc; close all; 

L = 0.01;                  % cavity side length [m]
nu = 1e-6;                 % kinematic viscosity [m^2/s]
N = 101;                   % grid points in each direction
nIter = 70000;             % original file used 70000
UpList = [0.01, 0.02, 0.05, 0.1, 1.0];

plotAfterEachCase = true;

% Time-step controls from the stability quantities:
% Courant number: Cr = U*dt/delta
% Diffusion number: D = nu*dt/delta^2
maxCourant = 1;
maxDiffusion = 0.5;

[x, y] = make_uniform_grid(N, L, L);
delta = x(2) - x(1);
[X, Y] = ndgrid(x, y);

omega = find_omega_psor(N, delta, L, X, Y);

fprintf('\nPSOR omega used in the solution: %.2f\n', omega);

results = [];

for caseNo = 1:length(UpList)
    Up = UpList(caseNo);
    Re = Up * L / nu;

    dt = choose_time_step(delta, Up, nu, maxCourant, maxDiffusion);

    fprintf('\nSolving Up = %.3g m/s   Re = %.0f   dt = %.6e s\n', Up, Re, dt);

    [zeta, psi, u, v, errorPsi, errorVel] = solve_cavity_case( ...
        N, Up, delta, dt, nu, omega, nIter);

    if plotAfterEachCase
       plot_streamlines_original_levels(x, y, psi, Re);
       drawnow
    end

    result = postprocess_case(psi, u, Up, Re, errorPsi, errorVel, x, y, delta);

    if caseNo == 1
        results = result;
    else
        results(caseNo) = result;
    end

    fprintf('Finished Re = %.0f   psi_min = %.6e   drag = %.6e\n', ...
        Re, result.psi_min_m2_s, result.Fdrag_N);
end

resultsTable = struct2table(results);

function [zeta, psi, u, v, errorPsi, errorVel] = solve_cavity_case( ...
    N, Up, delta, dt, nu, omega, nIter)

    psi = zeros(N, N);
    zeta = zeros(N, N);

    u = zeros(N, N);
    v = zeros(N, N);

    u(:, N) = Up;

    errorPsi = zeros(nIter, 1);
    errorVel = zeros(nIter, 1);

    for it = 1:nIter
        zeta = apply_vorticity_boundary(zeta, psi, delta, Up);

        zeta = adi_vorticity_original(u, v, zeta, dt, delta, delta, nu);

        psiOld = psi;
        psi = psor_streamfunction(omega, psi, 50, zeta, delta);

        uOld = u;
        vOld = v;
        [u, v] = velocity_from_streamfunction(psi, delta, u, v);

        errorPsi(it) = sum(sum(abs(psiOld - psi))) / (N - 2)^2;
        errorVel(it) = max(max(abs(uOld - u) + abs(vOld - v))) / Up;

        if mod(it, 5000) == 0
            fprintf('iter = %6d   errorPsi = %.3e   errorVel = %.3e\n', ...
                it, errorPsi(it), errorVel(it));
        end

        if ~isfinite(errorPsi(it)) || ~isfinite(errorVel(it))
            error('NaN or Inf appeared at iteration %d. The case diverged.', it)
        end
    end
end

function dt = choose_time_step(delta, Up, nu, maxCourant, maxDiffusion)

    dt_convection = maxCourant * delta / Up;
    dt_diffusion = maxDiffusion * delta^2 / nu;

    dt = min(dt_convection, dt_diffusion);
end

function zeta = apply_vorticity_boundary(zeta, psi, delta, Up)

    N = size(psi, 1);

    for k = 1:N
        zeta(1, k) = 2 * (psi(1, k) - psi(2, k)) / delta^2;
        zeta(N, k) = 2 * (psi(N, k) - psi(N-1, k)) / delta^2;

        zeta(k, 1) = 2 * (psi(k, 1) - psi(k, 2)) / delta^2;
        zeta(k, N) = 2 * (psi(k, N) - psi(k, N-1)) / delta^2 - 2 * Up / delta;
    end
end

function zeta = adi_vorticity_original(u, v, zeta, dt, deltaX, deltaY, nu)
    % ADI solution of the vorticity transport equation.
    % D_x = nu*dt/deltaX^2 and D_y = nu*dt/deltaY^2 are diffusion numbers.
    % epsx and epsy select the upwind direction.

    N = size(u, 1);

    Dx = nu * dt / deltaX^2;
    Dy = nu * dt / deltaY^2;

    a = zeros(N, 1);
    b = zeros(N, 1);
    c = zeros(N, 1);
    d = zeros(N, 1);

    % First sweep: x-direction implicit, y-direction explicit.
    for j = 2:N-1
        for i = 2:N-1
            epsx = sign_for_upwind(u(i,j));
            epsy = sign_for_upwind(v(i,j));

            a(i) = 0.5 * (Dx - 0.5 * (1 - epsx) * (u(i+1,j) * dt / deltaX));
            b(i) = 1 + Dx + 0.5 * epsx * (u(i,j) * dt / deltaX);
            c(i) = 0.5 * (Dx + 0.5 * (1 + epsx) * (u(i-1,j) * dt / deltaX));

            d(i) = ...
                0.5 * (Dy - 0.5 * (1 - epsy) * (v(i,j+1) * dt / deltaY)) * zeta(i,j+1) + ...
                (1 - Dy - 0.5 * epsy * (v(i,j) * dt / deltaY)) * zeta(i,j) + ...
                0.5 * (Dy + 0.5 * (1 + epsy) * (v(i,j-1) * dt / deltaY)) * zeta(i,j-1);
        end

        zeta(:,j) = thomas_ef(a, b, c, d, 0, zeta(1,j), zeta(N,j), zeta(:,j));
    end

    % Second sweep: y-direction implicit, x-direction explicit.
    for i = 2:N-1
        for j = 2:N-1
            epsx = sign_for_upwind(u(i,j));
            epsy = sign_for_upwind(v(i,j));

            a(j) = 0.5 * (Dy - 0.5 * (1 - epsy) * (v(i,j+1) * dt / deltaY));
            b(j) = 1 + Dy + 0.5 * epsy * (v(i,j) * dt / deltaY);
            c(j) = 0.5 * (Dy + 0.5 * (1 + epsy) * (v(i,j-1) * dt / deltaY));

            d(j) = ...
                0.5 * (Dx - 0.5 * (1 - epsx) * (u(i+1,j) * dt / deltaX)) * zeta(i+1,j) + ...
                (1 - Dx - 0.5 * epsx * (u(i,j) * dt / deltaX)) * zeta(i,j) + ...
                0.5 * (Dx + 0.5 * (1 + epsx) * (u(i-1,j) * dt / deltaX)) * zeta(i-1,j);
        end

        zeta(i,:) = thomas_ef(a, b, c, d, 0, zeta(i,1), zeta(i,N), zeta(i,:));
    end
end

function epsValue = sign_for_upwind(velocity)
    if velocity <= 0
        epsValue = -1;
    else
        epsValue = 1;
    end
end

function psi = psor_streamfunction(omega, psi, nSweeps, zeta, delta)
    % PSOR for:
    % psi_xx + psi_yy = -zeta
    % psi = psi + omega/4 * (neighbor sum - 4*psi + delta^2*zeta)

    N = size(psi, 1);

    for sweep = 1:nSweeps
        for i = 2:N-1
            for j = 2:N-1
                residual = ...
                    psi(i+1,j) + psi(i-1,j) + psi(i,j+1) + psi(i,j-1) - ...
                    4 * psi(i,j) + delta^2 * zeta(i,j);

                psi(i,j) = psi(i,j) + 0.25 * omega * residual;
            end
        end
    end
end

function [u, v] = velocity_from_streamfunction(psi, delta, u, v)
    % u = dpsi/dy
    % v = -dpsi/dx

    N = size(psi, 1);

    for i = 2:N-1
        for j = 2:N-1
            u(i,j) = (psi(i,j+1) - psi(i,j-1)) / (2 * delta);
            v(i,j) = -(psi(i+1,j) - psi(i-1,j)) / (2 * delta);
        end
    end
end

function q = thomas_ef(a, b, c, d, e1, f1, qN, q)
    % Recurrence:
    % q(i) = e(i)*q(i+1) + f(i)
    %
    % e1 and f1 impose the first boundary value.
    % qN imposes the last boundary value.

    N = length(a);

    e = zeros(N, 1);
    f = zeros(N, 1);

    q = q(:);

    e(1) = e1;
    f(1) = f1;

    for i = 2:N-1
        denominator = b(i) - c(i) * e(i-1);
        e(i) = a(i) / denominator;
        f(i) = (d(i) + c(i) * f(i-1)) / denominator;
    end

    q(N) = qN;

    for i = N-1:-1:1
        q(i) = e(i) * q(i+1) + f(i);
    end
end

function result = postprocess_case(psi, u, Up, Re, errorPsi, errorVel, x, y, delta)
    drag = drag_lid_original(u);

    psiMin = min(min(psi));
    [iCore, jCore] = find(psi == psiMin, 1);

    result.Up_m_s = Up;
    result.Re = Re;
    result.Fdrag_N = drag;
    result.psi_min_m2_s = psiMin;
    result.x_core_m = x(iCore);
    result.y_core_m = y(jCore);
    result.final_errorPsi = errorPsi(end);
    result.final_errorVel = errorVel(end);
end

function drag = drag_lid_original(u)

    mu = 1e-3;
    N = size(u, 1);

    u(1,N) = 0;
    u(N,N) = 0;

    drag = 0;

    for i = 1:N
        drag = drag + 0.5 * mu * (3*u(i,N) - 4*u(i,N-1) + u(i,N-2));
    end
end

function omega = find_omega_psor(N, delta, L, X, Y)

    zetaTest = sin(pi * X / L) .* sin(pi * Y / L);
    omegaValues = 1.85:0.01:1.99;

    bestError = inf;
    omega = omegaValues(1);
    previousError = inf;
    increaseCount = 0;

    for testOmega = omegaValues
        psiTest = zeros(N, N);

        for sweep = 1:200
            psiTest = psor_streamfunction(testOmega, psiTest, 1, zetaTest, delta);
        end

        residualSum = 0;

        for i = 2:N-1
            for j = 2:N-1
                residual = ...
                    psiTest(i+1,j) + psiTest(i-1,j) + psiTest(i,j+1) + psiTest(i,j-1) - ...
                    4 * psiTest(i,j) + delta^2 * zetaTest(i,j);

                residualSum = residualSum + abs(residual);
            end
        end

        currentError = residualSum / (N - 2)^2;
        fprintf('omega = %.2f   residual = %.6e\n', testOmega, currentError);

        if currentError < bestError
            bestError = currentError;
            omega = testOmega;
        end

        if currentError > previousError
            increaseCount = increaseCount + 1;
        else
            increaseCount = 0;
        end

        previousError = currentError;

        if increaseCount >= 2
            break
        end
    end
end

function plot_streamlines_original_levels(grid_x, grid_y, psi, Re)

N1 = 50;
N2 = 15;
diff1 = -min(min(psi));
dif1(1) = 1;
for i=2:N1
    dif1(i) = dif1(i-1) /exp(.02*i);
end
summy = sum(dif1);
dif1 = dif1/summy*diff1;
dif2(1) = 1;
for i=2:N2
    dif2(i) = dif2(i-1) /exp(.004*i);
end
summy = sum(dif2);
diff2 = max(max(psi));
dif2 = dif2/summy*diff2;
levels1(1) = min(min(psi));
for i=2:N1+1
    levels1(i) = levels1(i-1) + dif1(i-1);
end
levels2(1) = 0;
for i=2:N2+1
    levels2(i) = levels2(i-1) + dif2(i-1);
end

[Y, X] = meshgrid(grid_x, grid_y);
levels = [levels1, levels2, 2e-6, 1e-6, 2e-7, 5e-7, 1e-10, 1e-11, 1e-12, 1e-13, 1e-14, 1e-15, 1e-8, 1e-9];
figure;
set(gcf,'units','points','position',[100,100,670,600])
contour(X, Y, psi, levels, 'k');
title(sprintf('Cavity Streamlines, Re = %.0f', Re))
xlabel('x(m)');
ylabel('y(m)');
grid on

end

function [x, y] = make_uniform_grid(N, xLength, yLength)
    dx = xLength / (N - 1);
    dy = yLength / (N - 1);

    x = 0:dx:xLength;
    y = 0:dy:yLength;
end
